def _trim(s):
    a = s[:, :, 3]
    (ys, xs) = np.where(a > 0)
    if len(xs) == 0:

    return s[ys.min():ys.max() + 1, xs.min():xs.max() + 1]

    alive = merc_alive_slots(img)

        return []          # no gauge anywhere: window covered, loading, or the wrong screen
    last_alive = max(alive)
    out = []
    for slot in range(1, 13):
        if slot in alive:
            continue
        if slot < last_alive or slot in occupied:
            out.append(slot)
    return out

def fire_pixels(roi_bgr):
    if roi_bgr.size == 0:
        return 0
    b = roi_bgr[:, :, 0].astype(np.int16)
    g = roi_bgr[:, :, 1].astype(np.int16)
    r = roi_bgr[:, :, 2].astype(np.int16)
    orange = (r > FIRE_R_MIN) & (r - b > FIRE_RB_DIFF) & (g > FIRE_G_MIN)
    core = (r > FIRE_SEED_R) & (g > FIRE_SEED_G) & (b < FIRE_SEED_B_MAX)
    if not core.any():
        return 0
    m8 = cv2.morphologyEx((orange | core).astype(np.uint8), cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
    (n, lab) = cv2.connectedComponents(m8)
    if n <= 1:
        return 0
    valid = np.unique(lab[core])
    valid = valid[valid > 0]
    if len(valid) == 0:
        return 0
    return int(np.bincount(lab.ravel())[valid].max())

def _subtract_sprite_alpha(observed, reference):
    obs = observed.astype(np.float32)
    src = reference[:, :, :3].astype(np.float32)
    alpha = reference[:, :, 3:4].astype(np.float32) / 255.0
    remaining = np.maximum(1.0 - alpha, 0.00392156862745098)
    blended = (alpha[:, :, 0] > 0) & (alpha[:, :, 0] < 1) & np.all((obs >= alpha * src - 3) & (obs <= alpha * src + 255 * (1 - alpha) + 3), axis = 2)
    low = (obs - alpha * src - 3) / remaining
    high = ((obs - alpha * src) + 3) / remaining
    background = np.stack((high[:, :, 0], low[:, :, 1], low[:, :, 2]), axis = 2)
    filtered = observed.copy()
    filtered[blended] = np.clip(background, 0, 255).astype(np.uint8)[blended]
    same = (alpha[:, :, 0] > 0) & (np.abs(obs - src).max(axis = 2) <= 3)
    filtered[same] = 0
    return filtered

def _overlap_frac(mob, avoids):
    (mx1, my1, mx2, my2) = mob
    area = max(1.0, (mx2 - mx1) * (my2 - my1))
    best = 0.0
    for ax1, ay1, ax2, ay2 in avoids:
        iw = min(mx2, ax2) - max(mx1, ax1)
        ih = min(my2, ay2) - max(my1, ay1)
        if iw > 0 and ih > 0:
            best = max(best, iw * ih / area)
    return best

    repeated blacklisting with no fight in between (ESCALATE within ESCALATE_WINDOW) stops it.

    MIN_CLICKS = DUMMY_STACK_STOP
    MIN_SECONDS = 15.0
    STALE_SECONDS = 3.0
    BLACKLIST_SEC = 45.0
    BLACKLIST_RADIUS = 45.0
    ESCALATE = 3
    ESCALATE_WINDOW = 180.0
    # On a crowded map (many players standing on the mobs) most clicks land on a mob that cannot be
    # attacked, so several spots get skipped in a row. The rule therefore never stops the macro: once
    # enough spots are skipped it clears the whole list and starts over, which keeps it clicking until
    # the countdown actually starts. A real disconnect is caught by the separate no-fight clock.
    RESET_AFTER = 4

    def __init__(self):

        self.last_key = None
        self.last_t = None

    @property
    def clicks(self):
        """Re-clicks on the spot clicked last: the first click on a new spot is not a re-click."""
        v = self.spots.get(self.last_key)
        return v[0] - 1 if v else 0

    @property
    def first_t(self):
        v = self.spots.get(self.last_key)
        return v[1] if v else None

    def progress(self):
        """A fight started or the countdown digit was seen: the clicks worked."""
        self.spots = {}
        self.last_key = None
        self.last_t = None
        self.events = []

    def click(self, x, y, dist, now):
        """Register a mob click. Returns 'ok', 'suspect' (n-th unanswered re-click), 'skip' (skip this
        spot and keep hunting) or 'skip_all' (every skipped spot was cleared - retry everything).

        The counter is kept PER SPOT, not against the previous click: on a crowded map the nearest
        mob changes every time, and one shared counter that reset on every change never reached the
        limit, so nothing was ever skipped and the no-fight clock stopped the run.
        """
        if self.last_t is not None and now - self.last_t > self.STALE_SECONDS:
            self.spots = {}
            self.last_key = None
        self.last_t = now
        key = (int(round(x / 25.0)), int(round(y / 25.0)))
        self.last_key = key
        spot = self.spots.setdefault(key, [0, now])
        spot[0] += 1
        if spot[0] < self.MIN_CLICKS or now - spot[1] < self.MIN_SECONDS:
            return 'suspect' if spot[0] > 1 else 'ok'
        self.black.append((x, y, now + self.BLACKLIST_SEC))
        self.events = [t for t in self.events if now - t < self.ESCALATE_WINDOW] + [now]
        del self.spots[key]
        if len(self.events) >= self.ESCALATE and len(self.black) >= self.RESET_AFTER:
            self.black = []
            self.events = []
            self.spots = {}
            return 'skip_all'
        return 'skip'

    def filter(self, coords, now):
        """Drop detections on a skipped spot."""
        self.black = [b for b in self.black if b[2] > now]
        if coords is None or not self.black or coords.shape[0] == 0:
            return coords
        keep = np.ones(coords.shape[0], dtype=bool)
        for bx, by, _ in self.black:
            keep &= np.hypot(coords[:, 0] - bx, coords[:, 1] - by) > self.BLACKLIST_RADIUS
        return coords[keep]

SUMMON_MODES = ('none', 'taunt', 'non_taunt')

def summon_mode(settings: dict) -> str:
    mode = settings.get('summon_mode')
    if isinstance(mode, str) and mode in SUMMON_MODES:
        return mode
    return 'taunt' if settings.get('use_summon') is True else 'none'


def _sync_summon_mode(settings: dict):
    mode = summon_mode(settings)
    settings['summon_mode'] = mode
    settings['use_summon'] = mode != 'none'


def _load_raw() -> dict:
    
    try:
        with open(SETTINGS_PATH, encoding = 'utf-8') as f:
            return json.load(f)
    except Exception:
        return None


_cache_lock = threading.Lock()
_cache: 'dict | None' = None
_cache_mtime: 'float | None' = None

def _settings_mtime():
    
    try:
        return os.path.getmtime(SETTINGS_PATH)
    except OSError:
        return None



def load() -> dict:
    global _cache, _cache_mtime
    mt = _settings_mtime()
    with _cache_lock:
        if _cache and mt and mt == _cache_mtime:
            return copy.deepcopy(_cache)
    s = _load_raw() or {}
    monsters = s.setdefault('monsters', { })
    for name in MOB_CATALOG:
        monsters.setdefault(name, dict(DEFAULT_MONSTER_CFG))
    for mc in monsters.values():
        _sync_summon_mode(mc)
        for k, v in DEFAULT_MONSTER_CFG.items():
            mc.setdefault(k, v)
    s['monsters'] = dict(sorted(monsters.items()))
    for k, v in DEFAULT_COMMON.items():
        s.setdefault(k, v)
    if s.get('current_monster') not in MOB_CATALOG:
        s['current_monster'] = DEFAULT_MONSTER
    with _cache_lock:
        _cache = copy.deepcopy(s)
        _cache_mtime = mt
    return s


def save(s: dict) -> bool:
    global _cache, _cache_mtime
    tmp = None
    
    try:
        saved = copy.deepcopy(s)
        for mc in saved.get('monsters', { }).values():
            _sync_summon_mode(mc)
        data = json.dumps(saved, ensure_ascii = False, indent = 2)
        with _cache_lock:
            with tempfile.NamedTemporaryFile(mode = 'w', encoding = 'utf-8', dir = os.path.dirname(SETTINGS_PATH), prefix = '.prism_settings_', suffix = '.tmp', delete = False) as f:
                tmp = f.name
                f.write(data)
                f.flush()
                os.fsync(f.fileno())
            os.replace(tmp, SETTINGS_PATH)
            tmp = None
            _cache = saved
            _cache_mtime = _settings_mtime()
        if tmp:
            
            try:
                os.remove(tmp)
                return True
            except OSError:
                return True

        return True
    except Exception as e:
        logging.getLogger(__name__).warning('설정 저장 실패 (기존 설정 유지): %s', e)
        
        try:
            if tmp:
                
                try:
                    os.remove(tmp)
                    return False
                except OSError:
                    return False

            return False
        except:
            if tmp:
                os.remove(tmp)

class SettingsError(ValueError):
    pass


def _v_num(low, high):
    
    def rule(value, label):
        if isinstance(value, bool):
            raise SettingsError(f'''{label}: 숫자를 입력하세요''')
        
        try:
            out = float(value)
        except (TypeError, ValueError):
            raise SettingsError(f'''{label}: 숫자를 입력하세요''')

        if not (math.isfinite(out) and low <= out <= high):
            raise SettingsError(f'''{label}: {low:g}~{high:g} 범위여야 합니다''')
        return out

    return rule


def _v_bool(value, label):
    if not isinstance(value, bool):
        raise SettingsError(f'''{label}: 참/거짓 값이 필요합니다''')
    return value


def _v_summon_mode(value, label):
    if not isinstance(value, str) or value not in SUMMON_MODES:
        raise SettingsError(f'''{label}: 소환 사용 안 함 / 도발 탱 소환 / 비도발 탱 소환 중 선택하세요''')
    return value


def _v_squad(value, label):
    if not isinstance(value, list) or len(value) != 6:
        raise SettingsError(f'''{label}: 부대 1~6 값 6개가 필요합니다''')
    num = _v_num(0, 60)
    return [ num(v, label) for v in value ]


def _v_attack_keys(value, label):
    import combat
    return combat.normalize_attack_keys(value)


def _v_text(value, label):
    if value is None:
        value = ''
    return str(value).strip()


def _v_int(low, high):
    num = _v_num(low, high)
    
    def rule(value, label):
        return int(round(num(value, label)))

    return rule


def _v_choice(options):
    def rule(value, label):
        if value not in options:
            raise SettingsError(f'''{label}: {' / '.join(options)} 중 선택하세요''')
        return value

    return rule


def _v_waypoints(value, label):
    if not isinstance(value, list):
        raise SettingsError(f'''{label}: 기록된 경로 목록이어야 합니다''')
    out = []
    for step in value:
        if not isinstance(step, dict) or 'x' not in step or 'y' not in step:
            raise SettingsError(f'''{label}: 각 단계는 x, y 좌표가 필요합니다''')
        out.append({
            'x': int(step['x']),
            'y': int(step['y']),
            'wait': float(step.get('wait', 0.0))})
    return out

def _warn_missing_rules():
    log = logging.getLogger(__name__)
    for what, defaults, rules in (('몹 전투설정', DEFAULT_MONSTER_CFG, MONSTER_CFG_RULES), ('공통 설정', DEFAULT_COMMON, COMMON_RULES)):
        missing = set(defaults) - set(rules)
        if missing:
            log.warning('%s 검증 규칙 누락: %s (값은 보존하되 검증 안 함)', what, ', '.join(sorted(missing)))

_warn_missing_rules()

def _validate(raw, defaults, rules, base, what):
    if not isinstance(raw, dict):
        raise SettingsError(f'''{what} 형식이 올바르지 않습니다''')
    unknown = set(raw) - set(defaults)
    if unknown:
        raise SettingsError(f'''알 수 없는 {what}: ''' + ', '.join(sorted(unknown)))
    merged = dict(defaults)
    if isinstance(base, dict):
        merged.update({ k: v for k, v in base.items() if k in defaults })
    merged.update(raw)
    out = { }
    for key, default in defaults.items():
        rule = rules.get(key)
        value = merged.get(key, default)
        if rule:
            value = rule[1](value, rule[0])
        out[key] = copy.deepcopy(value)
    return out


def validate_monster_cfg(raw, base = None) -> dict:
    if isinstance(base, dict):
        base = dict(base)
        base['summon_mode'] = summon_mode(base)
    if isinstance(raw, dict) and 'summon_mode' not in raw and 'use_summon' in raw:
        raw = dict(raw)
        raw['summon_mode'] = 'taunt' if raw['use_summon'] is True else 'none'
    out = _validate(raw, DEFAULT_MONSTER_CFG, MONSTER_CFG_RULES, base, '전투 설정')
    _sync_summon_mode(out)
    return out


def validate_common(raw, base = None) -> dict:
    return _validate(raw, DEFAULT_COMMON, COMMON_RULES, base, '공통 설정')

The decompiled source counted an undecided block (None) as a rule violation, which the original

def sim(rnd, noise, drop, period=13.0, n_ops=15):
    ops, last = [], None
    for _ in range(n_ops):
        while True:
            a, b = sorted(rnd.sample(range(1, 6), 2))
            if last is None or not ({a, b} & set(last)):
                break
        ops.append((a, b))
        last = (a, b)
    n = int(period * n_ops) + 4
    pf = [None] * n
    for i, (a, b) in enumerate(ops):
        for f in range(int(round(period * i + 1)), min(n, int(round(period * (i + 1))))):
            u = rnd.random()
            if u < drop:
                continue
            pf[f] = tuple(sorted(rnd.sample(range(1, 6), 2))) if u < drop + noise else (a, b)
    return ops, pf
